import fs from 'node:fs';import assert from 'node:assert/strict';import {randomUUID} from 'node:crypto';import {spawn} from 'node:child_process';
const root=process.cwd(),{docker}=await import(root+'/scripts/runner-process.mjs'),{outputCapture}=await import(root+'/scripts/runner-output.mjs'),{browserContainerOptions}=await import(root+'/scripts/browser-container.mjs');
const images=[['after',fs.readFileSync('/tmp/refactor-runtime-image-complete.txt','utf8').trim()]],reports=[];
for(const [label,image] of images){const id=randomUUID(),name='zombie-browser-held-'+id,marker='ZOMBIE_REACT_'+id,start=performance.now();
try{const created=await docker(['create',...browserContainerOptions(name),'-i',image,'timeout','-s','KILL','14','node','/opt/zombie-browser/runner.mjs']);assert.equal(created.exitCode,0,created.stderr);
const result=await new Promise((resolve,reject)=>{const child=spawn('docker',['start','--attach','--interactive',name],{stdio:['pipe','pipe','pipe']}),capture=outputCapture(262144);const timer=setTimeout(()=>{child.kill('SIGKILL');void docker(['rm','--force',name]);},17000);child.stdout.on('data',b=>capture.push('out',b));child.stderr.on('data',b=>capture.push('err',b));child.stdin.on('error',()=>{});child.on('error',reject);child.on('close',exitCode=>{clearTimeout(timer);resolve({...capture.result(),exitCode});});
// Intentionally withhold EOF. Both images receive the same complete JSON line.
child.stdin.write(JSON.stringify({language:'typescript',environment:'react',code:'export default ()=> <p>fixed public probe</p>',steps:[],marker})+'\n');});
let observed;try{observed=JSON.parse(result.stdout);}catch{}
const report={label,image,exitCode:result.exitCode,durationMs:Math.round(performance.now()-start),executed:result.stderr.includes(marker),sandboxVerified:observed?.sandboxVerified??false,phase:observed?.phase??null,domText:observed?.domText??null,stdoutEmpty:result.stdout.length===0,stdinEOF:false};reports.push(report);console.log(JSON.stringify(report));
if(label==='before'){assert.equal(report.executed,false);assert.equal(report.exitCode,137);assert.equal(report.stdoutEmpty,true);}else{assert.equal(report.executed,true);assert.equal(report.exitCode,0);assert.equal(report.sandboxVerified,true);assert.equal(report.domText,'fixed public probe');}
}finally{await docker(['rm','--force',name]);}}
fs.writeFileSync('/tmp/refactor-react-held-stdin-complete.json',JSON.stringify(reports,null,2)+'\n');
